import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

from sklearn.preprocessing import LabelEncoder, MinMaxScaler

pd.set_option("display.max_columns", None)
pd.set_option("display.float_format", "{:.2f}".format)

df = pd.read_csv("Titanic-Dataset.csv")

display(df.head())

display(df.head())
display(df.tail())

print("Rows and Columns:", df.shape)
print("Number of Rows:", df.shape[0])
print("Number of Columns:", df.shape[1])

print(df.columns.tolist())

df.info()

display(df.describe())

display(df.describe(include="object"))


display(df.isnull().sum())


missing_count = df.isnull().sum()
missing_percentage = (df.isnull().mean() * 100).round(2)

missing_summary = pd.DataFrame({
    "Missing Count": missing_count,
    "Missing Percentage": missing_percentage
})

display(missing_summary)

df.isnull().sum().plot(kind="bar", figsize=(10, 5))

plt.title("Missing Values in Each Column")
plt.xlabel("Columns")
plt.ylabel("Missing Count")
plt.xticks(rotation=45)
plt.show()


plt.figure(figsize=(10, 5))

sns.heatmap(df.isnull(), cbar=False, cmap="viridis")

plt.title("Missing Values Heatmap")
plt.show()



missing_count = df.isnull().sum()
missing_count = missing_count[missing_count > 0]

missing_count.plot(kind="pie", autopct="%1.1f%%", figsize=(7, 7))

plt.title("Distribution of Missing Values")
plt.ylabel("")
plt.show()



display(df.dtypes)


numerical_columns = [
    "PassengerId", "Survived", "Pclass",
    "Age", "SibSp", "Parch", "Fare"
]

categorical_columns = [
    "Name", "Sex", "Ticket", "Cabin", "Embarked"
]

print("Numerical Columns:", numerical_columns)
print("Categorical Columns:", categorical_columns)



df.columns = df.columns.str.strip().str.lower().str.replace(" ", "_")

print(df.columns.tolist())


df["name"] = df["name"].str.strip().str.lower()
df["sex"] = df["sex"].str.strip().str.lower()
df["ticket"] = df["ticket"].str.strip().str.lower()
df["embarked"] = df["embarked"].str.strip().str.lower()

display(df.isnull().sum())

df["age"] = df["age"].fillna(df["age"].median())

df["embarked"] = df["embarked"].fillna(
    df["embarked"].mode()[0]
)

df = df.drop(columns=["cabin"])

display(df.isnull().sum())


df_processed = df.copy()

df_processed["passengerid"] = df_processed["passengerid"].astype(str)
df_processed["survived"] = df_processed["survived"].astype(int)
df_processed["pclass"] = df_processed["pclass"].astype("category")

display(df_processed.dtypes)


label_encoder = LabelEncoder()

df_processed["sex"] = label_encoder.fit_transform(
    df_processed["sex"]
)

display(df_processed[["sex"]].head())


print(dict(zip(
    label_encoder.classes_,
    label_encoder.transform(label_encoder.classes_)
)))

One hot Encoding

df_processed = pd.get_dummies(
    df_processed,
    columns=["embarked"],
    prefix="embarked",
    dtype=int
)

display(df_processed.head())


display(df_processed.head())
display(df_processed.dtypes)


display(df_processed[["age", "fare"]].head(10))


scaler = MinMaxScaler()

df_processed[["age", "fare"]] = scaler.fit_transform(
    df_processed[["age", "fare"]]
)

display(df_processed[["age", "fare"]].head(10))


print("Minimum Values:")
display(df_processed[["age", "fare"]].min())

print("Maximum Values:")
display(df_processed[["age", "fare"]].max())


print("Final Dataset Shape:", df_processed.shape)

df_processed.info()

display(df_processed.head())


display(df_processed.isnull().sum())

print("Total Missing Values:", df_processed.isnull().sum().sum())


df_processed.to_csv("Titanic_Preprocessed.csv", index=False)

print("Preprocessed dataset saved successfully!")
