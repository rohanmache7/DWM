np.random.seed(42)

n = 1000

income = np.random.randint(20000, 150001, n)
credit_score = np.random.randint(450, 851, n)

employment_status = np.random.choice(
    ["Employed", "Self-Employed", "Unemployed"],
    size=n,
    p=[0.65, 0.25, 0.10]
)

loan_amount = np.random.randint(50000, 1000001, n)

loan_term = np.random.choice(
    [12, 24, 36, 48, 60],
    size=n
)

past_default = np.random.choice(
    ["Yes", "No"],
    size=n,
    p=[0.20, 0.80]
)

# Create an underlying approval score
score = (
    0.000015 * income
    + 0.012 * credit_score
    - 0.0000025 * loan_amount
    - 0.01 * loan_term
    + np.where(employment_status == "Employed", 1.2, 0)
    + np.where(employment_status == "Self-Employed", 0.7, 0)
    + np.where(employment_status == "Unemployed", -1.5, 0)
    + np.where(past_default == "No", 1.5, -2.0)
)

# Add randomness
score += np.random.normal(0, 1.5, n)

loan_approved = (score > np.median(score)).astype(int)

df = pd.DataFrame({
    "Income": income,
    "Credit_Score": credit_score,
    "Employment_Status": employment_status,
    "Loan_Amount": loan_amount,
    "Loan_Term": loan_term,
    "Past_Default": past_default,
    "Loan_Approved": loan_approved
})
